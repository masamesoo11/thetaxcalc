   149→function SalaryCard({ amount }: { amount: number }) {
   150→  const calc = calculateSalaryTakeHome(amount);
   151→  // Texas (0% state tax) = highest net
   152→  const txNet = calc.states.find((s) => s.stateKey === 'texas')!.netAnnual;
   153→
   154→  return (
   155→    <Link
   156→      href={`/salary/${salaryToSlug(amount)}`}
   157→      className="group premium-card hover-lift p-5 flex flex-col gap-3"
   158→    >
   159→      {/* Header */}
   160→      <div className="flex items-center justify-between">
   161→        <h3 className="text-xl font-bold text-foreground group-hover:text-emerald-400 transition-colors">
   162→          {formatSalary(amount)}
   163→        </h3>
   164→        <span className="text-[10px] font-semibold uppercase tracking-wider rounded-full px-2.5 py-1 bg-emerald-500/10 text-emerald-400 border border-emerald-500/20">
   165→          {formatSalaryCompact(amount)}
   166→        </span>
   167→      </div>
   168→
   169→      {/* Texas take-home (highest net) */}
   170→      <div className="flex items-center gap-2">
   171→        <MapPin className="h-4 w-4 text-emerald-400 shrink-0" />
   172→        <span className="text-sm text-muted-foreground">
   173→          TX Take-Home: <strong className="text-emerald-400">{fmt(txNet)}</strong>/yr
   174→        </span>
   175→      </div>
   176→
   177→      {/* Mini state comparison */}
   178→      <div className="grid grid-cols-5 gap-1 mt-1">
   179→        {calc.states.map((state) => {
   180→          const pct = state.netAnnual / txNet;
   181→          return (
   182→            <div key={state.stateKey} className="text-center">
   183→              <div className="h-1.5 rounded-full bg-muted/30 overflow-hidden">
   184→                <div
   185→                  className="h-full rounded-full bg-emerald-500/60"
   186→                  style={{ width: `${pct * 100}%` }}
   187→                />
   188→              </div>
   189→              <p className="text-[10px] text-muted-foreground mt-1">{state.stateAbbr}</p>
   190→            </div>
   191→          );
   192→        })}
   193→      </div>
   194→
   195→      {/* CTA */}
   196→      <div className="mt-auto flex items-center gap-1.5 text-sm font-medium text-emerald-400 group-hover:gap-3 transition-all">
   197→        View Details

   199→      </div>
   200→    </Link>
   201→  );
